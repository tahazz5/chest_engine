cmake_minimum_required(VERSION 3.20)
project(chess_engine VERSION 0.5.0 LANGUAGES CXX)

add_library(chess_core STATIC src/board/position.cpp src/board/attacks.cpp src/movegen/movegen.cpp)
target_include_directories(chess_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(chess_core PUBLIC cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHESS_DISABLE_PEXT "Disable hardware PEXT even on supported CPUs" OFF)
if(CHESS_DISABLE_PEXT)
    target_compile_definitions(chess_core PRIVATE CHESS_DISABLE_PEXT=1)
endif()

function(chess_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
chess_warnings(chess_core)

add_library(chess_play STATIC src/play/play.cpp)
target_link_libraries(chess_play PUBLIC chess_core)
chess_warnings(chess_play)
add_executable(chess_engine src/main.cpp)
target_link_libraries(chess_engine PRIVATE chess_play)
chess_warnings(chess_engine)

include(CTest)
if(BUILD_TESTING)
    add_executable(chess_tests tests/core_tests.cpp)
    target_link_libraries(chess_tests PRIVATE chess_core)
    chess_warnings(chess_tests)
    foreach(suite IN ITEMS types squares moves bitboards)
        add_test(NAME ${suite} COMMAND chess_tests ${suite})
    endforeach()
    add_executable(position_tests tests/position_tests.cpp)
    target_link_libraries(position_tests PRIVATE chess_core)
    chess_warnings(position_tests)
    foreach(suite IN ITEMS initial roundtrip invalid invariants lifetime)
        add_test(NAME position_${suite} COMMAND position_tests ${suite})
    endforeach()
    add_executable(attack_tests tests/attack_tests.cpp)
    target_link_libraries(attack_tests PRIVATE chess_core)
    chess_warnings(attack_tests)
    foreach(suite IN ITEMS leapers rays tables queries)
        add_test(NAME attacks_${suite} COMMAND attack_tests ${suite})
    endforeach()
    add_executable(movegen_tests tests/movegen_tests.cpp)
    target_link_libraries(movegen_tests PRIVATE chess_core)
    chess_warnings(movegen_tests)
    foreach(suite IN ITEMS list initial pawns castling oracle boundary)
        add_test(NAME movegen_${suite} COMMAND movegen_tests ${suite})
    endforeach()
    add_executable(legal_tests tests/legal_tests.cpp)
    target_link_libraries(legal_tests PRIVATE chess_core)
    chess_warnings(legal_tests)
    foreach(suite IN ITEMS perft undo legality)
        add_test(NAME legal_${suite} COMMAND legal_tests ${suite})
    endforeach()
    add_executable(play_tests tests/play_tests.cpp)
    target_link_libraries(play_tests PRIVATE chess_play)
    chess_warnings(play_tests)
    foreach(suite IN ITEMS interaction endings opponent)
        add_test(NAME play_${suite} COMMAND play_tests ${suite})
    endforeach()
endif()

option(CHESS_BUILD_BENCHMARKS "Build standalone attack microbenchmark" OFF)
if(CHESS_BUILD_BENCHMARKS)
    add_executable(attack_bench benchmark/attack_bench.cpp)
    target_link_libraries(attack_bench PRIVATE chess_core)
    chess_warnings(attack_bench)
endif()
